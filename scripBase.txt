INSERT INTO detalle_presupuesto (cantidad, descricion, precio_unitario, sub_total, id_presupuesto)
SELECT 
    d.cantidad,
    d.trabajo_realizado AS descricion,
    -- Asignación de precio unitario estimado según el tipo de ítem
    CASE 
        WHEN d.tipo_item = 'SERVICIO' THEN 55000.00
        WHEN d.tipo_item = 'MANO_OBRA' THEN 35000.00
        WHEN d.tipo_item = 'REPUESTO' THEN 45000.00
        ELSE 25000.00
    END AS precio_unitario,
    -- Cálculo del subtotal: cantidad * precio_unitario
    (d.cantidad * CASE 
        WHEN d.tipo_item = 'SERVICIO' THEN 55000.00
        WHEN d.tipo_item = 'MANO_OBRA' THEN 35000.00
        WHEN d.tipo_item = 'REPUESTO' THEN 45000.00
        ELSE 25000.00
    END) AS sub_total,
    p.id_presupuesto
FROM detalle_ordenes d
JOIN orden_trabajo o ON d.id_asignacion = o.id_asignacion
JOIN presupuesto p ON o.id_asignacion = p.id_orden;


UPDATE presupuesto p
JOIN orden_trabajo o ON p.id_orden = o.id_asignacion
JOIN turno t ON o.id_turno = t.id_turno
SET p.id_vehiculo = t.id_vehiculo
WHERE p.id_vehiculo IS NULL;

UPDATE presupuesto 
SET id_administrador = 1 
WHERE id_administrador IS NULL;
